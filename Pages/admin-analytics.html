import { supabase } from "./api.js"; document.addEventListener("DOMContentLoaded", () => { loadFullAnalytics(); }); async function loadFullAnalytics() { try { // 1. Fetch Orders and Feedbacks from Supabase const { data: orders } = await supabase.from('orders').select('*'); const { data: feedbacks } = await supabase.from('feedback').select('*'); const ordersList = orders || []; const feedbacksList = feedbacks || []; // 2. Parse Timestamps const now = new Date(); const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()); const todayOrders = ordersList.filter(o => new Date(o.created_at) >= startOfToday); const completedOrders = ordersList.filter(o => o.status === "DELIVERED" || o.status === "delivered"); const pendingOrders = ordersList.filter(o => o.status === "pending" || o.status === "PREPARING"); // 3. Calculate Financials const totalRevenue = completedOrders.reduce((sum, o) => sum + (Number(o.total_amount) || 0), 0); // 4. Calculate Average Rating let avgRating = 5.0; if (feedbacksList.length > 0) { const sumRatings = feedbacksList.reduce((sum, f) => sum + (Number(f.rating) || 5), 0); avgRating = (sumRatings / feedbacksList.length).toFixed(1); } // 5. Update Metrics Cards updateDOM("stat-total-orders", todayOrders.length); updateDOM("stat-completed-orders", completedOrders.length); updateDOM("stat-pending-orders", pendingOrders.length); updateDOM("stat-revenue", `₹${totalRevenue}`); updateDOM("stat-avg-rating", `⭐ ${avgRating} / 5.0`); renderRecentFeedbacks(feedbacksList); } catch (err) { console.error("Failed to load dashboard metrics:", err); } } function renderRecentFeedbacks(feedbacks) { const container = document.getElementById("admin-feedback-feed"); if (!container) return; if (feedbacks.length === 0) { container.innerHTML = `

No customer reviews yet.

`; return; } container.innerHTML = feedbacks.slice(0, 5).map(f => `
Order #${f.order_id || 'Direct'} ${"★".repeat(f.rating || 5)}

"${f.comment || 'No comment'}"

`).join(""); } function updateDOM(id, val) { const el = document.getElementById(id); if (el) el.innerText = val; }
